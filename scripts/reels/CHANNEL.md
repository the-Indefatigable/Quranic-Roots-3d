# QuRoots channel kit

    python scripts/reels/engine/reel_audio.py scripts/reels/specs/hamd.json out/hamd
    node scripts/reels/render.mjs scripts/reels/engine/reel.html out/hamd.mp4 <duration> out/hamd/soundtrack.wav out/hamd/cues.json scripts/reels/specs/hamd.json
